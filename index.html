      }`;


      const badgeColor = g.type === 'pauta' ? 'bg-pauta/20 text-pauta border-pauta/30' : 'bg-organico/20 text-organico border-organico/30';

      const labelText = g.type === 'pauta' ? 'PAUTA' : 'ORGÁNICO';


      div.innerHTML = `

        <div class="flex justify-between items-start mb-3">

          <span class="px-2.5 py-0.5 rounded-md text-[10px] font-black tracking-wider uppercase border ${badgeColor}">${labelText}</span>

          <button onclick="copyScriptText('${g.id}')" class="text-xs text-gray-400 hover:text-white p-1 rounded bg-black/40 hover:bg-black/60 border border-white/10" title="Copiar texto">

            <i class="fa-regular fa-copy"></i>

          </button>

        </div>

        <h3 class="font-extrabold text-white text-base mb-2 group-hover:text-blue-400 transition-colors">${g.title}</h3>

        <p class="text-xs text-gray-400 mb-4 line-clamp-2">${g.hook}</p>

        <div class="flex justify-between items-center pt-3 border-t border-white/5 text-xs">

          <button onclick="openModal('${g.id}')" class="text-blue-400 hover:underline font-bold flex items-center gap-1">

            <i class="fa-solid fa-expand text-[10px]"></i> Leer Guion

          </button>

          <span class="text-[10px] text-gray-500"><i class="fa-solid fa-grip-dots-vertical"></i> Mover</span>

        </div>

      `;

      return div;

    }


    function allowDrop(ev) { ev.preventDefault(); }

    function drag(ev) { ev.dataTransfer.setData("text", ev.target.id); }


    function drop(ev, targetId) {

      ev.preventDefault();

      const data = ev.dataTransfer.getData("text");

      const card = document.getElementById(data);

      const target = document.getElementById(targetId);


      if (target) {

        const placeholder = target.querySelector('p');

        if (placeholder) placeholder.remove();

        target.appendChild(card);

        updateCounts();

      }

    }


    function updateCounts() {

      document.getElementById('count-sesion1').innerText = `${document.getElementById('sesion1').querySelectorAll('.glass-card').length} guiones`;

      document.getElementById('count-sesion2').innerText = `${document.getElementById('sesion2').querySelectorAll('.glass-card').length} guiones`;

    }


    function openModal(id) {

      const g = guiones.find(item => item.id === id);

      if (!g) return;


      const modalCategory = document.getElementById('modalCategory');

      modalCategory.innerText = g.type === 'pauta' ? 'PAUTA' : 'ORGÁNICO';

      modalCategory.className = `px-3 py-1 rounded-full text-xs font-bold uppercase ${

        g.type === 'pauta' ? 'bg-pauta/20 text-pauta border border-pauta/30' : 'bg-organico/20 text-organico border border-organico/30'

      }`;


      document.getElementById('modalTitle').innerText = g.title;

      document.getElementById('modalHook').innerText = g.hook;

      document.getElementById('modalBody').innerText = g.text;

      document.getElementById('scriptModal').classList.remove('hidden');

      document.getElementById('scriptModal').classList.add('flex');

    }


    function closeModal() {

      document.getElementById('scriptModal').classList.add('hidden');

      document.getElementById('scriptModal').classList.remove('flex');

    }


    function copyModalText() {

      const text = document.getElementById('modalBody').innerText;

      navigator.clipboard.writeText(text);

      alert('¡Guion copiado al portapapeles!');

    }


    function copyScriptText(id) {

      const g = guiones.find(item => item.id === id);

      if (g) {

        navigator.clipboard.writeText(g.text);

        alert(`¡Texto de "${g.title}" copiado!`);

      }

    }


    function setFilter(filter) {

      currentFilter = filter;

      document.querySelectorAll('[id^="btn-"]').forEach(btn => btn.className = btn.className.replace('bg-white/20', 'bg-white/5'));

      document.getElementById(`btn-${filter}`).className += ' bg-white/20';

      filterCards();

    }


    function filterCards() {

      const query = document.getElementById('searchInput').value.toLowerCase();

      guiones.forEach(g => {

        const card = document.getElementById(g.id);

        if (!card) return;

        const matchesSearch = g.title.toLowerCase().includes(query) || g.text.toLowerCase().includes(query) || g.hook.toLowerCase().includes(query);

        const matchesFilter = currentFilter === 'all' || g.type === currentFilter;


        if (matchesSearch && matchesFilter) {

          card.classList.remove('hidden');

        } else {

          card.classList.add('hidden');

        }

      });

    }


    document.addEventListener('DOMContentLoaded', renderCards);
